def findMaximum(nums):
    if len(nums) == 0:
        return "List is empty"

    nums.sort()          # Efficient sorting (Timsort - O(n log n))
    return nums[-1]      # Maximum element


# Test Case 1
nums = []
print(findMaximum(nums))

# Test Case 2
nums = [5]
print(findMaximum(nums))

# Test Case 3
nums = [3, 3, 3, 3, 3]
print(findMaximum(nums))
