def isPalindrome(s):
    return s == s[::-1]

def firstPalindrome(words):
    for word in words:
        if isPalindrome(word):
            return word
    return ""

# Example 1
words = ["abc", "car", "ada", "racecar", "cool"]
print(firstPalindrome(words))

# Example 2
words = ["notapalindrome", "racecar"]
print(firstPalindrome(words))
